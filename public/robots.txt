# Все краулеры — одна группа, с одними правилами.
#
# Так нужно, а не просто короче. Краулер слушает ровно одну группу — ту, где
# назван он сам, — и общую «User-agent: *» при этом не читает (RFC 9309).
# До 06.10.2026 у каждого названного краулера была своя группа с одним
# «Allow: /», и запрет служебных фрагментов, стоявший только в общей группе,
# действовал на одного Googlebot. Яндекс, Bing и ИИ-краулеры видели
# фрагменты как дубликаты статей.
#
# Внутри группы нет пустых строк намеренно: старые разборщики считают
# пустую строку концом группы. Комментарии внутри группы разрешены.

# --- ИИ-краулеры ---------------------------------------------------------
# Разрешены намеренно. Большинство сайтов их сейчас закрывает; мы делаем
# обратное: проект существует ради того, чтобы его читали, в том числе
# через разговор с моделью.
User-agent: GPTBot
User-agent: ChatGPT-User
User-agent: OAI-SearchBot

User-agent: Claude-SearchBot

User-agent: YandexBot
User-agent: Bingbot
# --- Все остальные, включая Googlebot -------------------------------------
User-agent: *
Allow: /
# Служебные фрагменты, которые подгружает читалка. Это куски статей без
# оболочки — для читателя они бессмысленны, для поисковика выглядят
# дубликатом статьи.
Disallow: /posts/*/partial
